```python
# controllers/portal.py
@http.route('/my/guide/loans/new', type='http', auth='user', methods=['GET', 'POST'])
def portal_loan_new(self, **kw):
    guide = self._get_current_guide()
    if request.httprequest.method == 'POST':
        # 创建借款申请
        loan = request.env['tour.guide.loan'].sudo().create({
            'guide_id': guide.id,
            'employee_id': guide.employee_id.id,
            'amount': float(kw['amount']),
            'purpose': kw['purpose'],
            'plan_id': int(kw['plan_id']) if kw.get('plan_id') else False,
            'state': 'draft',
        })
        return request.redirect('/my/guide/loans/%s' % loan.id)
    # GET: 返回申请表单
```

```python
# models/tour_guide_loan.py
@api.model
def _cron_loan_reminder(self):
    """借款到期提醒"""
    today = fields.Date.today()
    for days in [7, 3, 1]:
        due_date = fields.Date.add(today, days=days)
        loans = self.search([
            ('state', '=', 'disbursed'),
            ('due_date', '=', due_date),
        ])
        for loan in loans:
            # 发送提醒
            loan._send_reminder_notification(days)
```

```python
# controllers/portal.py
@http.route('/my/guide/expenses/statistics', type='http', auth='user', website=True)
def portal_expense_statistics(self, **kw):
    guide = self._get_current_guide()
    expenses = request.env['hr.expense'].search([
        ('x_guide_id', '=', guide.id),
    ])
    # 按月份统计
    monthly_stats = {}
    for expense in expenses:
        month = expense.date.strftime('%Y-%m')
        monthly_stats[month] = monthly_stats.get(month, 0) + expense.total_amount
    
    return {'monthly_stats': monthly_stats}
```

#### 4.2 收入汇总
**现状**: 无收入汇总功能

- 年度收入汇总
- 按来源分类（佣金/报账/其他）
- 收入趋势图

**实现方案**:
```python
# controllers/portal.py
@http.route('/my/guide/income_summary', type='http', auth='user', website=True)
def portal_income_summary(self, year=None, **kw):
    guide = self._get_current_guide()
    if not year:
        year = fields.Date.today().year
    
    # 佣金收入
    commissions = request.env['tour.commission'].search([
        ('guide_id', '=', guide.id),
        ('state', '=', 'paid'),
        ('pay_date', '>=', f'{year}-01-01'),
        ('pay_date', '<=', f'{year}-12-31'),
    ])
    commission_total = sum(commissions.mapped('guide_net_pay'))
    
    # 报账收入
    expenses = request.env['hr.expense'].search([
        ('x_guide_id', '=', guide.id),
        ('state', '=', 'paid'),
        ('date', '>=', f'{year}-01-01'),
        ('date', '<=', f'{year}-12-31'),
    ])
    expense_total = sum(expenses.mapped('total_amount'))
    
    return {
        'year': year,
        'commission_total': commission_total,
        'expense_total': expense_total,
        'total': commission_total + expense_total,
    }
```

## 技术实现建议

### 1. 安全性
- 所有财务数据需要严格的权限控制
- 敏感信息（银行账号）需要脱敏显示
- 导出功能需要记录日志

### 2. 性能优化
- 大数据量查询使用分页
- 统计报表使用缓存
- 附件上传限制大小和类型

### 3. 用户体验
- 使用图表展示统计数据
- 提供搜索和筛选功能
- 移动端适配

## 总结

当前导游门户已实现基础的财务查询和报账提交功能，但在以下方面需要完善：

1. **数据导出** - 佣金/收入证明导出
2. **流程跟踪** - 审批进度可视化
3. **附件管理** - 发票/收据上传
4. **统计分析** - 收入汇总和趋势
5. **消息通知** - 财务变动提醒
6. **自助服务** - 借款申请、账户管理

建议按优先级逐步实现，优先解决基础功能缺失问题，再完善高级功能。
